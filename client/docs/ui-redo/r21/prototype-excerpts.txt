 function toggle(label,id,checked,description=''){return `<label class="preference-row"><span>${T(label)}${description?`<small>${T(description)}</small>`:''}</span><input type="checkbox" role="switch" id="${id}" ${checked?'checked':''}></label>`;}
 function confirm(title,message,action,label='Confirm',danger=false){const close=showPaymentDialog(`<h2 id="payment-title">${T(title)}</h2><p>${T(message)}</p><button id="action-confirm" class="primary ${danger?'danger':''}">${T(label)}</button><button id="action-cancel" class="text-button">${T('Cancel')}</button>`);$('#action-cancel').onclick=close;$('#action-confirm').onclick=()=>{close();action();};}
 function settings(){const settingRow=(label,id,glyph,value='')=>`<button class="menu-row settings-icon-row" id="${id}">${icon(glyph)}<span class="settings-row-label">${T(label)}</span><small>${escape(value)}</small><span class="settings-row-arrow" aria-hidden="true">${icon('chevron')}</span></button>`;page('Settings',`<div class="menu-group">${settingRow('Account & security','account-security','lock')}${settingRow('Privacy','privacy-settings','shield')}${settingRow('Notifications','notification-settings','bell')}</div><div class="menu-group">${settingRow('General','general-settings','gear')}${settingRow('Language','language-settings','globe',window.MauGoI18n.get()==='ko'?'한국어':'English')}</div><div class="menu-group">${settingRow('Feedback','feedback-settings','feedback')}${settingRow('Help & support','help-settings','help')}${settingRow('About MauGo','about-settings','info')}</div><button class="text-button settings-signout" id="logout">${T('Sign out')}</button><button class="delete-account-link" id="delete-account-settings">${T('Delete account')}</button>`);view.classList.add('settings-reference');for(const[id,name]of Object.entries({'account-security':'security','privacy-settings':'privacy','notification-settings':'notifications','language-settings':'language','general-settings':'general','feedback-settings':'feedback','help-settings':'help','about-settings':'about'}))$('#'+id).onclick=()=>open(name);$('#delete-account-settings').onclick=()=>open('delete-account');$('#logout').onclick=()=>confirm('Sign out?','Your local letters and messages will be kept for this preview profile.',()=>window.MauGoAuth.logout(),'Sign out');}
 function language(){state.settings.language=window.MauGoI18n.get();delete state.settings.quiet;page('Language',`<p class="intro-copy">${T('Choose the language that feels like home.')}</p><div class="language-options"><button data-language="en" aria-pressed="${state.settings.language==='en'}"><span>English</span><b>${state.settings.language==='en'?'✓':''}</b></button><button data-language="ko" aria-pressed="${state.settings.language==='ko'}"><span>한국어</span><b>${state.settings.language==='ko'?'✓':''}</b></button></div>`);view.querySelectorAll('[data-language]').forEach(b=>b.onclick=()=>{state.settings.language=b.dataset.language;save('maugo-language',state.settings.language);persist();window.MauGoI18n.set(state.settings.language);render();});}
 function privacy(){page('Privacy',`<div class="menu-group">${row('Profile visibility','profile-visibility',T(state.settings.profileVisibility))}${toggle('Show online status','online-status',state.settings.online)}${toggle('Show visit history','visit-history',state.settings.visits)}${toggle('Privacy screen','privacy-screen',state.settings.privacyScreen,'Blur the app when you switch away.')}${row('Blocked people','blocked-people')}</div><p class="local-note">${T('Privacy screen does not prevent screenshots. Visibility settings are saved locally until a server is connected.')}</p>`);bindToggles({'online-status':'online','visit-history':'visits','privacy-screen':'privacyScreen'});$('#profile-visibility').onclick=()=>chooseVisibility(state.settings.profileVisibility,value=>{state.settings.profileVisibility=value;persist();render();});$('#blocked-people').onclick=()=>open('blocked');}
 function chooseVisibility(current,done){showSheet('Who can see this?',`<div class="visibility-options">${['Everyone','Friends','Only me'].map(v=>`<button data-visibility="${v}" aria-pressed="${current===v}">${T(v)}<b>${current===v?'✓':''}</b></button>`).join('')}</div>`);$('#sheet-content').querySelectorAll('[data-visibility]').forEach(b=>b.onclick=()=>{closeSheet();done(b.dataset.visibility);});}
 function notifications(){page('Notifications',`<div class="menu-group">${toggle('Allow app notifications','allow-notifications',state.settings.notifications)}${toggle('Replies','reply-notifications',state.settings.replyNotifications)}${toggle('Home reply banners','home-reply-banners',state.settings.homeReplyBanners)}${toggle('Likes','like-notifications',state.settings.likeNotifications)}${toggle('New followers','follower-notifications',state.settings.followerNotifications)}${toggle('Sound','notification-sound',state.settings.sound)}</div><p class="local-note">${T('Notification preferences are saved on this device. Background push requires the native app service.')}</p>`);bindToggles({'allow-notifications':'notifications','reply-notifications':'replyNotifications','home-reply-banners':'homeReplyBanners','like-notifications':'likeNotifications','follower-notifications':'followerNotifications','notification-sound':'sound'});for(const id of ['reply-notifications','home-reply-banners','like-notifications','follower-notifications','notification-sound'])$('#'+id).disabled=!state.settings.notifications;const change=$('#allow-notifications').onchange;$('#allow-notifications').onchange=e=>{change(e);render();};}
 function general(){page('General',`<div class="menu-group">${toggle('Haptics','haptics',state.settings.haptics)}${toggle('Personalized recommendations','recommendations',state.settings.recommendations)}${toggle('Use system font','system-font',state.settings.systemFont)}${row('Clear temporary cache','clear-cache')}</div><p class="local-note">${T('Your letters, messages and photos will be kept.')}</p>`);bindToggles({'haptics':'haptics','recommendations':'recommendations','system-font':'systemFont'});$('#clear-cache').onclick=()=>confirm('Clear temporary cache?','Your letters, messages and photos will be kept.',async()=>{if(false){const keys=await caches.keys();await Promise.all(keys.filter(k=>k.startsWith('maugo-')).map(k=>caches.delete(k)));}toast(T('Temporary cache cleared.'));});}
